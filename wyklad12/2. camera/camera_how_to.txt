#include "camera.hpp"
CFreeCamera myCamera;

// 2. Inicjalizacja w Initialize()
// obecnie domyslne ustawienia obiektu sa ustawiane 
// w domyslnym konstruktorze klasy CCamera, ale dobrym
// pomyslem jest uzupelnienie metody Init() w klasie 
// CFreeCamera wg potrzeby, obecnie niepotrzebne

myCamera.Init(windowWidth, windowHeight);


// 3. W funkcji zwrotnej obslugujacej zmiane rozdzielczosci
// Reshape() / framebuffer_size_callback()
// nalezy zaktualizowac macierz rzutowania 
// metoda UpdatePerspective()

myCamera.UpdatePerspective(screen_width, screen_height);

// 4. W DisplayScene() przesylamy aktualne macierze kamery
// matProj oraz matView do potoku

myCamera.SetUniformPV(idProgram, "matProj", "matView");

// 5. Skorzystanie z metody CFreeCamera::Rotate() do zmiany
// orientacji kamery w funkcji cursor_position_callback()

void cursor_position_callback(GLFWwindow* window, double xpos, double ypos)
{
	if (__mouse_press && __mouse_button == GLFW_MOUSE_BUTTON_LEFT)
	{
		float dx = 2.0*(xpos - __mouse_buttonX)/(float)windowHeight;
		float dy = 2.0*(ypos - __mouse_buttonY)/(float)windowWidth;
		__mouse_buttonX = xpos;
		__mouse_buttonY = ypos;

		myCamera.Rotate(dx, dy);
	}
}
